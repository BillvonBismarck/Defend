    clear() { this.hitboxes.length = 0; },
    hit(x, y, w, h, cb, label) { this.hitboxes.push({ x, y, w, h, cb, label }); },
    click(x, y) {
      for (let i = this.hitboxes.length - 1; i >= 0; i--) {
        const h = this.hitboxes[i];
        if (x >= h.x && x <= h.x + h.w && y >= h.y && y <= h.y + h.h) { h.cb(); return true; }
      }
      return false;
    },
    button(ctx, x, y, w, h, text, cb, opt = {}) {
      ctx.save();
      roundRect(ctx, x, y, w, h, opt.r || 18, opt.fill || "rgba(0,216,255,0.12)", opt.stroke || C.cyan, 2);
      ctx.fillStyle = opt.color || C.white;
      ctx.font = `${opt.size || 26}px Microsoft YaHei, Arial`;
      ctx.textAlign = "center";
      ctx.textBaseline = "middle";
      ctx.fillText(text, x + w / 2, y + h / 2);
      ctx.restore();
      this.hit(x, y, w, h, cb, text);
    }
  };

  function roundRect(ctx, x, y, w, h, r, fill, stroke, line = 1) {
    ctx.beginPath();
    ctx.moveTo(x + r, y);
    ctx.lineTo(x + w - r, y);
    ctx.quadraticCurveTo(x + w, y, x + w, y + r);
    ctx.lineTo(x + w, y + h - r);
    ctx.quadraticCurveTo(x + w, y + h, x + w - r, y + h);
    ctx.lineTo(x + r, y + h);
    ctx.quadraticCurveTo(x, y + h, x, y + h - r);
    ctx.lineTo(x, y + r);
    ctx.quadraticCurveTo(x, y, x + r, y);
    if (fill) { ctx.fillStyle = fill; ctx.fill(); }
    if (stroke) { ctx.strokeStyle = stroke; ctx.lineWidth = line; ctx.stroke(); }
  }

  function drawText(ctx, text, x, y, size = 24, color = C.white, align = "left") {
    ctx.fillStyle = color;
    ctx.font = `${size}px Microsoft YaHei, PingFang SC, Arial`;
    ctx.textAlign = align;
    ctx.textBaseline = "middle";
    ctx.fillText(text, x, y);
  }

  const Player = {
    init(charId) {
      this.char = CHARACTERS.find(c => c.id === charId) || CHARACTERS[0];
      this.x = WORLD_SIZE / 2;
      this.y = WORLD_SIZE / 2;
      this.level = 1;
      this.exp = 0;
      this.expToNext = CONFIG.LEVEL.expFormula(1);
      this.kills = 0;
      this.skills = {};
      this.invTimer = 0;
      this.attackTimer = 0;
      this.overloadKills = 0;
      this.overloadActive = false;
      this.overloadTimer = 0;
      this.shieldCharges = 0;
      this.shieldTimer = 0;
      this.magnetBoostTimer = 0;
      this.killSpeedTimer = 0;
      this.radius = CONFIG.PLAYER.radius;
      this.resetBonuses();
      this.recalc();
      DailyChallenge.applyToPlayer();
      RuneResonance.apply();
      this.hp = this.maxHp;
    },
    resetBonuses() {
      this.atkBonus = 0; this.hpBonus = 0; this.speedBonus = 0; this.critBonus = 0; this.magnetBonus = 0;
      this.fireRateBonus = 0; this.extraBullets = 0; this.pierceCount = 0; this.bounceCount = 0;
      this.knockback = 0; this.bigBulletChance = 0; this.homingChance = 0; this.lootBonus = 0;
      this.regenRate = 0; this.dodgeChance = 0; this.comboBonus = 0; this.killSpeedBonus = 0; this.shieldIntervalBonus = 0;
    },
    applyMetaBonuses() {
      const r = Save.data.relics, t = Save.data.totems, ru = Save.data.runes;
      this.atkBonus += r.power * 0.04;
      this.hpBonus += r.life * 0.05;
      this.lootBonus += r.luck * 0.04;
      this.magnetBonus += r.magnet * 0.06;
      this.metaHpFlat = t.hp * 2;
      this.metaAtkFlat = t.atk * 0.8;
      this.metaSpeedFlat = t.speed * 8;
      this.runeVampire = ru.vampire * 0.03;
      this.runeKing = ru.king * 0.02;
      if (Save.data.selectedSkin === "gold") this.lootBonus += 0.1;
      if (Save.data.selectedSkin === "dark") this.critBonus += 0.05;
    },
    recalc() {
      const P = CONFIG.PLAYER, L = CONFIG.LEVEL, s = this.char.stats;
      const oldMax = this.maxHp || 1;
      const hpRatio = this.hp ? this.hp / oldMax : 1;
      this.metaHpFlat = 0; this.metaAtkFlat = 0; this.metaSpeedFlat = 0; this.runeVampire = 0; this.runeKing = 0;
      const savedSkills = { ...this.skills };
      this.resetBonuses();
      this.skills = savedSkills;
      this.applyMetaBonuses();
      for (const id in this.skills) {
        const def = SkillSystem.def(id);
        if (def && def.passive) for (let i = 0; i < this.skills[id]; i++) def.passive(this);
      }
      const lvl = this.level;
      const all = 1 + this.runeKing;
      this.maxHp = Math.floor((P.baseHp * s.hp + this.metaHpFlat) * (1 + L.hpGrowth * (lvl - 1)) * (1 + this.hpBonus) * all);
      this.atk = Math.floor((P.baseAtk * s.atk + this.metaAtkFlat) * (1 + L.atkGrowth * (lvl - 1)) * (1 + this.atkBonus) * all);
      this.speed = (P.baseSpeed * s.speed + this.metaSpeedFlat) * (1 + this.speedBonus) * all;
      this.critRate = P.baseCritRate * s.critRate + L.critGrowth * (lvl - 1) + this.critBonus;
      this.attackInterval = Math.max(0.1, P.attackInterval * (1 - this.fireRateBonus));
      if (this.hp !== undefined) this.hp = clamp(Math.floor(this.maxHp * hpRatio), 1, this.maxHp);
    },
    update(dt, mx, my) {
      let speed = this.speed * SpecialTerrain.speedMult();
      if (this.overloadActive) speed *= 1 + CONFIG.OVERLOAD.speedBonus;
      if (this.killSpeedTimer > 0) speed *= 1 + this.killSpeedBonus * 4;
      this.x = clamp(this.x + mx * speed * dt, 40, WORLD_SIZE - 40);
      this.y = clamp(this.y + my * speed * dt, 40, WORLD_SIZE - 40);
      this.attackTimer += dt;
      this.invTimer = Math.max(0, this.invTimer - dt);
      this.shieldTimer += dt;
      this.magnetBoostTimer = Math.max(0, this.magnetBoostTimer - dt);
      this.killSpeedTimer = Math.max(0, this.killSpeedTimer - dt);
      if (this.regenRate > 0) this.heal(this.maxHp * this.regenRate * dt);
      if (this.skills.shield) {
        const interval = Math.max(4, 11 - this.skills.shield * 1.5 - this.shieldIntervalBonus * 10);
        if (this.shieldTimer >= interval) {
          this.shieldTimer = 0;
          this.shieldCharges = Math.min(this.skills.shield, this.shieldCharges + 1);
        }
      }
      if (this.overloadActive) {
        this.overloadTimer -= dt;
        if (this.overloadTimer <= 0) this.overloadActive = false;
      }
    },
    effectiveAtk() {
      let a = this.atk;
      if (this.overloadActive) a *= CONFIG.OVERLOAD.atkMultiplier;
      return Math.max(1, Math.floor(a));
    },
    effectiveInterval() {
      let it = this.attackInterval;
      if (this.overloadActive) it *= CONFIG.OVERLOAD.cooldownReduction;
      if (this.comboBonus) it *= Math.max(0.45, 1 - Math.min(0.35, this.comboBonus * Math.min(10, this.kills % 10) / 10));
      return it;
    },
    damage(amount) {
      if (this.invTimer > 0 || BattleScene.state !== BATTLE.PLAYING) return false;
      if (Math.random() < this.dodgeChance) { FX.floatText(this.x, this.y - 20, "闪避", C.cyan, 24); this.invTimer = 0.25; return false; }
      if (this.shieldCharges > 0) { this.shieldCharges--; this.invTimer = 0.35; FX.floatText(this.x, this.y - 20, "护盾", C.cyan, 24); return false; }
      const armor = Math.min(0.45, Save.data.relics.life * 0.015);
      const dmg = Math.max(1, Math.floor(amount * (1 - armor)));
      if (SpecialTerrain.invincible()) { FX.floatText(this.x, this.y - 20, "无敌", C.yellow, 22); return false; }
      this.hp -= dmg;
      this.invTimer = CONFIG.PLAYER.invincibleTime;
      FX.floatText(this.x, this.y - 26, "-" + dmg, C.red, 24);
      if (this.hp <= 0) BattleScene.finish(false);
      return true;
    },
    heal(v) { this.hp = Math.min(this.maxHp, this.hp + v); },
    addExp(v) {
      this.exp += v;
      let up = false;
      while (this.exp >= this.expToNext) {
        this.exp -= this.expToNext;
        this.level++;
        this.expToNext = CONFIG.LEVEL.expFormula(this.level);
        this.recalc();
        this.hp = Math.min(this.maxHp, this.hp + Math.max(3, Math.floor(this.maxHp * 0.15)));
        up = true;
      }
      return up;
    },
    addKill(enemy) {
      this.kills++;
      this.overloadKills = Math.min(CONFIG.OVERLOAD.killsToFull, this.overloadKills + 1);
      if (this.overloadKills >= CONFIG.OVERLOAD.killsToFull && !this.overloadActive) {
        this.overloadActive = true;
        this.overloadTimer = CONFIG.OVERLOAD.duration;
        this.overloadKills = 0;
      }
      if (this.killSpeedBonus) this.killSpeedTimer = 1.5;
      if (this.runeVampire && Math.random() < this.runeVampire) this.heal(this.maxHp * 0.04);
      Achievement.check();
    }
  };

  const Wave = {
    reset() { this.waveNum = 0; this.timer = 0; this.totalTime = 0; this.waveCoeff = 1; this.bossAlive = false; this.spawnTimer = 0; this.gameOver = false; this.lastGiftWave = -99; },
    pickEnemyType() {
      const w = this.waveNum;
      let pool;
      if (w <= 2) pool = [["normal", 1]];
      else if (w <= 4) pool = [["normal", .65], ["fast", .35]];
      else if (w <= 6) pool = [["normal", .30], ["fast", .20], ["swarm", .18], ["tank", .15], ["ranger", .17]];
      else if (w <= 9) pool = [["normal", .20], ["fast", .17], ["swarm", .13], ["tank", .10], ["charger", .13], ["ghost", .12], ["ranger", .15]];
      else if (w <= 12) pool = [["normal", .12], ["fast", .12], ["swarm", .10], ["tank", .10], ["charger", .12], ["ghost", .10], ["shaman", .12], ["ranger", .12], ["elite", .10]];
      else pool = [["normal", .08], ["fast", .10], ["swarm", .09], ["tank", .09], ["charger", .11], ["ghost", .11], ["shaman", .11], ["ranger", .13], ["elite", .18]];
      let r = Math.random(), a = 0;
      for (const e of pool) { a += e[1]; if (r <= a) return e[0]; }
      return pool[pool.length - 1][0];
    },
    update(dt) {
      this.totalTime += dt;
      this.spawnTimer += dt;
      const diff = CONFIG.DIFFICULTY[Save.data.difficulty];
      let spawnInterval = this.waveNum <= 30 ? Math.max(0.8, 2 - this.waveNum * 0.05) : Math.max(0.3, 0.8 - (this.waveNum - 30) * 0.025);
      const normalHp = ENEMY_DEFS.normal.hp * this.waveCoeff;
      if (Player.atk * 2 > normalHp) spawnInterval *= 0.35;
      if (this.spawnTimer >= spawnInterval) {
        this.spawnTimer = 0;
        if (Enemy.active.length < CONFIG.WAVE.maxEnemiesAlive * 0.5) {
          const type = this.pickEnemyType();
          const count = type === "swarm" ? randInt(2, 3) : 1;
          for (let i = 0; i < count; i++) Enemy.spawnAround(Player.x, Player.y, type, this.waveCoeff);
        }
      }
      this.timer += dt;
      if (this.timer >= CONFIG.WAVE.interval) {
        this.timer = 0;
        this.waveNum++;
        let baseCoeff = 1 + CONFIG.WAVE.coeffGrowth * this.waveNum;
        if (this.waveNum >= 50) baseCoeff *= 2 * Math.pow(1.01, this.waveNum - 50);
        else if (this.waveNum >= 40) baseCoeff *= 2;
        else if (this.waveNum >= 30) baseCoeff *= Math.pow(1.03, this.waveNum - 30);
        this.waveCoeff = baseCoeff * diff.enemyHpMult;
        let count = Math.floor((CONFIG.WAVE.baseEnemyCount + CONFIG.WAVE.countGrowth * this.waveNum) * diff.spawnMult);
        for (let i = 0; i < count; i++) Enemy.spawnAround(Player.x, Player.y, this.pickEnemyType(), this.waveCoeff);
        Enemy.spawnAround(Player.x, Player.y, "elite", this.waveCoeff);
        if (!this.bossAlive && this.waveNum % CONFIG.WAVE.bossInterval === 0) {
          const idx = Math.floor(this.waveNum / CONFIG.WAVE.bossInterval) - 1;
          const bossDef = BOSSES[idx % BOSSES.length];
          Enemy.spawnBoss(Player.x, Player.y, bossDef, this.waveCoeff);
          this.bossAlive = true;
        }
        Achievement.check();
      }
    }
  };

  const Enemy = {
    active: [], nextId: 1,
    reset() { this.active.length = 0; this.nextId = 1; },
    spawnAround(px, py, type, coeff) {
      const a = Math.random() * TAU, d = rand(CONFIG.WAVE.spawnMinRadius, CONFIG.WAVE.spawnRadius);
      this.spawn(clamp(px + Math.cos(a) * d, 50, WORLD_SIZE - 50), clamp(py + Math.sin(a) * d, 50, WORLD_SIZE - 50), type, coeff);
    },
    spawn(x, y, type, coeff) {
      const def = ENEMY_DEFS[type] || ENEMY_DEFS.normal;
      const diff = CONFIG.DIFFICULTY[Save.data.difficulty];
      const e = {
        id: this.nextId++, type, emoji: def.emoji, name: def.name, color: def.color,
        x, y, hp: Math.floor(def.hp * coeff), maxHp: Math.floor(def.hp * coeff), atk: Math.floor(def.atk * coeff * diff.enemyAtkMult),
        speed: def.speed * diff.enemySpeedMult, radius: def.radius, expDrop: def.exp, alive: true,
        isBoss: false, shootTimer: rand(0, 2), dashTimer: rand(2, 4), dashVX: 0, dashVY: 0, dashTime: 0, slowTimer: 0, charmed: false, charmHitTimer: 0
      };
      Save.data.codex.enemies[type] = true;
      this.active.push(e);
    },
    spawnBoss(px, py, bossDef, coeff) {
      const a = Math.random() * TAU, d = 620;
      const e = {
        id: this.nextId++, type: "boss", emoji: bossDef.emoji, name: bossDef.name, color: bossDef.color,
        x: clamp(px + Math.cos(a) * d, 80, WORLD_SIZE - 80), y: clamp(py + Math.sin(a) * d, 80, WORLD_SIZE - 80),
        hp: Math.floor(bossDef.hp * Math.max(1, coeff * 0.65)), maxHp: Math.floor(bossDef.hp * Math.max(1, coeff * 0.65)), atk: Math.floor(12 * coeff),
        speed: 75, radius: 38, expDrop: 120, alive: true, isBoss: true, bossDef, skillTimers: [1.5, 3.5, 7], stunTimer: 0, dashTime: 0, dashVX: 0, dashVY: 0
      };
      Save.data.codex.bosses[bossDef.id] = true;
      this.active.push(e);
      FX.banner(`${bossDef.name} 出现`);
    },
    nearest(x, y, range, includeBoss = true) {
      let best = null, bd = range * range;
      for (const e of this.active) {
        if (!e.alive || e.charmed || (!includeBoss && e.isBoss)) continue;
        const d = distSq(x, y, e.x, e.y);
        if (d < bd) { bd = d; best = e; }
      }
      return best;
    },
    nearestToEnemy(src, range) {
      let best = null, bd = range * range;
      for (const e of this.active) {
        if (!e.alive || e === src || e.charmed) continue;
        const d = distSq(src.x, src.y, e.x, e.y);
        if (d < bd) { bd = d; best = e; }
      }
      return best;
    },
    damage(e, dmg, color = C.white) {
      if (!e.alive) return false;
      dmg = Math.max(1, Math.floor(dmg));
      DamageStats.add("攻击/技能", dmg);
      e.hp -= dmg;
      FX.floatText(e.x, e.y - e.radius, String(Math.floor(dmg)), color, e.isBoss ? 26 : 18);
      if (e.hp <= 0) { this.kill(e); return true; }
      return false;
    },
    kill(e) {
      if (!e.alive) return;
      e.alive = false;
      Player.addKill(e);
      Loot.spawnFromEnemy(e.x, e.y, e.expDrop, e.isBoss);
      FX.burst(e.x, e.y, e.color, e.isBoss ? 28 : 10);
      if (e.isBoss) {
        Wave.bossAlive = false;
        if (BOSSES[BOSSES.length - 1].id === e.bossDef.id) BattleScene.finish(true);
      }
    },
    update(dt) {
      for (const e of this.active) {
        if (!e.alive) continue;
        if (e.slowTimer > 0) e.slowTimer -= dt;
        const speedMul = e.slowTimer > 0 ? 0.45 : 1;

        if (e.charmed) {
          const t = this.nearestToEnemy(e, 900);
          if (t) {
            const [nx, ny] = norm(t.x - e.x, t.y - e.y);
            e.x += nx * e.speed * 1.25 * dt;
            e.y += ny * e.speed * 1.25 * dt;
            e.charmHitTimer = (e.charmHitTimer || 0) + dt;
            if (e.charmHitTimer > 0.4 && distSq(e.x, e.y, t.x, t.y) < (e.radius + t.radius + 12) ** 2) {
              e.charmHitTimer = 0;
              this.damage(t, Player.effectiveAtk() * 0.8, C.pink);
            }
          }
          continue;
        }

        if (e.dashTime > 0) {
          e.dashTime -= dt;
          e.x += e.dashVX * dt; e.y += e.dashVY * dt;
        } else {
          const dx = Player.x - e.x, dy = Player.y - e.y;
          const [nx, ny] = norm(dx, dy);
          if (e.type === "charger") {
            e.dashTimer -= dt;
            if (e.dashTimer <= 0) { e.dashTimer = rand(3.2, 5); e.dashTime = 0.45; e.dashVX = nx * 420; e.dashVY = ny * 420; }
          }
          if (e.type === "ghost") {
            e.x += nx * e.speed * speedMul * dt * (0.7 + 0.3 * Math.sin(Wave.totalTime * 4 + e.id));
            e.y += ny * e.speed * speedMul * dt * (0.7 + 0.3 * Math.cos(Wave.totalTime * 3 + e.id));
          } else {
            e.x += nx * e.speed * speedMul * dt;
            e.y += ny * e.speed * speedMul * dt;
          }
        }

        if (e.type === "ranger" || e.type === "shaman") {
          e.shootTimer -= dt;
          if (e.shootTimer <= 0) {
            e.shootTimer = e.type === "shaman" ? 3.2 : 2.4;
            EnemyBullet.spawnTarget(e.x, e.y, Player.x, Player.y, e.type === "shaman" ? 180 : 240, e.atk, e.type === "shaman" ? "🔮" : "➹", e.color);
          }
        }

        if (e.isBoss) this.updateBoss(e, dt);

        if (!e.isBoss && distSq(e.x, e.y, Player.x, Player.y) < (e.radius + Player.radius) ** 2) {
          Player.damage(e.atk);
        }
      }
      this.active = this.active.filter(e => e.alive);
    },
    updateBoss(e, dt) {
      if (e.stunTimer > 0) { e.stunTimer -= dt; return; }
      for (let i = 0; i < e.skillTimers.length; i++) {
        e.skillTimers[i] -= dt;
        if (e.skillTimers[i] <= 0) {
          e.skillTimers[i] = i === 0 ? 3.0 : i === 1 ? 5.5 : 12.0;
          this.castBossSkill(e, e.bossDef.skills[i], i === 2);
          if (i === 2) e.stunTimer = 2.2;
        }
      }
    },
    castBossSkill(e, type, ultimate) {
      FX.floatText(e.x, e.y - 50, ultimate ? "大招" : e.name, e.color, 22);
      const dmg = Math.floor(e.atk * (ultimate ? 1.5 : 1));
      if (type === "shoot_fan") EnemyBullet.spawnFan(e.x, e.y, Player.x, Player.y, 7, 0.9, 250, dmg, "●", e.color);
      else if (type === "shoot_burst") for (let i = 0; i < 4; i++) setTimeout(() => EnemyBullet.spawnTarget(e.x, e.y, Player.x, Player.y, 230, dmg, "◆", e.color), i * 140);
      else if (type === "shoot_target") EnemyBullet.spawnTarget(e.x, e.y, Player.x, Player.y, 190, dmg + 4, "⬢", e.color, 15);
      else if (type === "shoot_radial") EnemyBullet.spawnRadial(e.x, e.y, ultimate ? 24 : 14, 210, dmg, "●", e.color);
      else if (type === "shoot_split") EnemyBullet.spawnSplit(e.x, e.y, Player.x, Player.y, 180, dmg, e.color);
      else if (type === "dash") { const [nx, ny] = norm(Player.x - e.x, Player.y - e.y); e.dashTime = 0.55; e.dashVX = nx * 480; e.dashVY = ny * 480; }
      else if (type === "area_rain") for (let i = 0; i < 12; i++) EnemyBullet.spawnArea(Player.x + rand(-260, 260), Player.y + rand(-360, 360), 42, dmg, 0.9, e.color);
      else if (type === "area_circle") EnemyBullet.spawnArea(Player.x, Player.y, ultimate ? 150 : 95, dmg + 3, 1.1, e.color);
      else if (type === "blink_attack") { e.x = clamp(Player.x + rand(-180, 180), 80, WORLD_SIZE - 80); e.y = clamp(Player.y + rand(-180, 180), 80, WORLD_SIZE - 80); EnemyBullet.spawnRadial(e.x, e.y, 10, 190, dmg, "✦", e.color); }
    },
    killNonBoss() { for (const e of this.active) if (!e.isBoss) this.kill(e); }
  };

  const Projectile = {
    active: [],
    reset() { this.active.length = 0; },
    spawn(x, y, tx, ty, dmg, pierce = 0, bounce = 0, opt = {}) {
      const [nx, ny] = norm(tx - x, ty - y);
      const big = opt.big || Math.random() < Player.bigBulletChance;
      this.active.push({
        x, y, vx: nx * CONFIG.BULLET.speed, vy: ny * CONFIG.BULLET.speed, dmg,
        radius: CONFIG.BULLET.radius * (big ? 1.8 : 1), lifetime: CONFIG.BULLET.lifetime + (pierce + bounce) * 0.18,
        pierce, bounce, hit: new Set(), color: opt.color || C.cyan, emoji: opt.emoji || (big ? "🔵" : "•"), homing: opt.homing || Math.random() < Player.homingChance, alive: true
      });
    },
    spawnSpread(x, y, tx, ty, dmg, pierce, extra, bounce) {
      const base = Math.atan2(ty - y, tx - x);
      const total = extra + 1;
      const spread = Math.min(0.9, 0.18 * extra);
      for (let i = 0; i < total; i++) {
        const a = base + (i - (total - 1) / 2) * (spread / Math.max(1, total - 1));
        this.spawn(x, y, x + Math.cos(a) * 100, y + Math.sin(a) * 100, dmg, pierce, bounce, { emoji: "•" });
      }
    },
    update(dt) {
      const kills = [];
      for (const p of this.active) {
        if (!p.alive) continue;
        if (p.homing) {
          const t = Enemy.nearest(p.x, p.y, 240);
          if (t) {
            const [nx, ny] = norm(t.x - p.x, t.y - p.y);
            p.vx = p.vx * 0.92 + nx * CONFIG.BULLET.speed * 0.08;
            p.vy = p.vy * 0.92 + ny * CONFIG.BULLET.speed * 0.08;
          }
        }
        p.x += p.vx * dt; p.y += p.vy * dt; p.lifetime -= dt;
        if (p.lifetime <= 0 || p.x < -100 || p.x > WORLD_SIZE + 100 || p.y < -100 || p.y > WORLD_SIZE + 100) { p.alive = false; continue; }
        for (const e of Enemy.active) {
          if (!e.alive || e.charmed || p.hit.has(e.id)) continue;
          if (distSq(p.x, p.y, e.x, e.y) <= (p.radius + e.radius) ** 2) {
            p.hit.add(e.id);
            let damage = p.dmg;
            const crit = Math.random() < Player.critRate;
            if (crit) damage *= CONFIG.PLAYER.critDamage;
            if (Enemy.damage(e, damage, crit ? C.yellow : p.color)) kills.push(e);
            if (p.bounce > 0) {
              p.bounce--;
              const t = Enemy.nearest(p.x, p.y, 360);
              if (t) { const [nx, ny] = norm(t.x - p.x, t.y - p.y); p.vx = nx * CONFIG.BULLET.speed; p.vy = ny * CONFIG.BULLET.speed; }
            } else if (p.pierce > 0) p.pierce--;
            else { p.alive = false; break; }
          }
        }
      }
      this.active = this.active.filter(p => p.alive);
      return kills;
    }
  };

      Enemy.update(dt);
      EnemyBullet.update(dt);
      this.handlePickups(Loot.update(dt));
      FX.update(dt);
      this.updateCamera();
    },
    autoAttack() {
      const interval = Player.effectiveInterval();
      if (Player.attackTimer < interval) return;
      const target = Enemy.nearest(Player.x, Player.y, CONFIG.PLAYER.attackRange);
      if (!target) return;
      Player.attackTimer = 0;
      const damage = Math.floor(Player.effectiveAtk() * SpecialTerrain.atkMult());
      if (Player.extraBullets > 0) Projectile.spawnSpread(Player.x, Player.y, target.x, target.y, damage, Player.pierceCount, Player.extraBullets, Player.bounceCount);
      else Projectile.spawn(Player.x, Player.y, target.x, target.y, damage, Player.pierceCount, Player.bounceCount);
    },
    handleKills(kills) {
      if (!kills || !kills.length) return;
      Achievement.check();
    },
    handlePickups(items) {
      for (const it of items) {
        if (it.type === "exp") {
          if (Player.addExp(it.value)) {
            this.skillChoices = SkillSystem.randomChoices();
            if (this.skillChoices.length) this.state = BATTLE.SKILL_SELECT;
          }
        } else if (it.type === "heart") Player.heal(Player.maxHp * 0.15);
        else if (it.type === "magnet") Loot.bigMagnet = 5;
        else if (it.type === "bomb") Enemy.killNonBoss();
        else if (it.type === "coin") { this.sessionGold += it.value; FX.floatText(Player.x, Player.y - 40, "+金币", C.yellow, 20); }
        else if (it.type === "gift") this.state = BATTLE.GIFT_AD;
      }
    },

      const tx = clamp(Player.x - DESIGN_W / 2, 0, WORLD_SIZE - DESIGN_W);
      const ty = clamp(Player.y - DESIGN_H / 2, 0, WORLD_SIZE - DESIGN_H);

      this.camY += (ty - this.camY) * 0.1;

    }
